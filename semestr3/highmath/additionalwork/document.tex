\textbf{Задача №1.1. Дослідити на збіжність числовий ряд}
\[
\sum_{n=0}^{\infty}\frac{n+2}{\sqrt{n^2+1}}.
\]

Розглянемо загальний член ряду:
\[
a_n=\frac{n+2}{\sqrt{n^2+1}}.
\]

Перевіримо необхідну умову збіжності:
\[
\lim_{n\to\infty}a_n
=\lim_{n\to\infty}\frac{n+2}{\sqrt{n^2+1}}
=\lim_{n\to\infty}\frac{1+\frac{2}{n}}{\sqrt{1+\frac{1}{n^2}}}
=1.
\]

Оскільки \(\lim a_n\neq0\), ряд не може бути збіжним.

\vspace{1em}

\textbf{Задача №1.2. Дослідити на збіжність числовий ряд}
\[
\sum_{n=1}^{\infty}\frac{\arctg\!\left(\frac{1+(-1)^n}{2}n\right)}{n^3+2}.
\]

Тут видно, що:
\[
\frac{1+(-1)^n}{2}=
\begin{cases}
1, & n\ \text{парне},\\
0, & n\ \text{непарне}.
\end{cases}
\]

Тоді:
\[
0\le a_n\le\frac{\pi/2}{n^3}.
\]

Оскільки ряд
\[
\sum_{n=1}^{\infty}\frac{1}{n^3}
\]
є збіжним, то за 1 ознакою порівняння даний ряд також збіжний (можна сказати, що абсолютно збіжний).

\vspace{1em}

\textbf{Задача №1.7. Дослідити на збіжність числовий ряд}
\[
\sum_{n=3}^{\infty}\frac{(-1)^n}{(n+1)\ln n}.
\]

Розглянемо ряд із модулів:
\[
\sum_{n=3}^{\infty}\frac{1}{(n+1)\ln n}.
\]

Порівняємо з відомим рядом:
\[
\sum_{n=3}^{\infty}\frac{1}{n\ln n},
\]
який є розбіжним.

Оскільки
\[
\frac{1}{(n+1)\ln n}\sim\frac{1}{n\ln n},
\]
то ряд з модулів розбіжний.

\vspace{1em}

Позначимо:
\[
a_n=\frac{1}{(n+1)\ln n}.
\]

Маємо:
\[
a_n>0,\qquad a_n\searrow,\qquad \lim_{n\to\infty}a_n=0.
\]

Отже, за ознакою Лейбніца ряд збіжний. Звідси робимо висновок, що ряд є умовно збіжним.

\vspace{1em}

\textbf{Задача №1.10. Дослідити на збіжність ряд}
\[
\sum_{n=1}^{\infty}\frac{(x-5)^{2n+1}}{3n+8}.
\]


Розглянемо ряд із модулів:
\[
\sum_{n=1}^{\infty}\frac{|x-5|^{2n+1}}{3n+8}
=|x-5|\sum_{n=1}^{\infty}\frac{|x-5|^{2n}}{3n+8}.
\]

Застосуємо ознаку Д’Аламбера:
\[
\lim_{n\to\infty}\frac{|x-5|^{2(n+1)}}{|x-5|^{2n}}=|x-5|^2.
\]

Ряд абсолютно збіжний при:
\[
|x-5|^2<1\quad\Rightarrow\quad|x-5|<1.
\]

\textbf{Перевірка кінців.}

\[
x=4 (\text{ або }) 6\Rightarrow\sum\frac{1}{3n+8}\ \text{розбіжний}.
\]

\textbf{Висновок:} область збіжності (4; 6).

\vspace{1em}

\textbf{Задача №3.2. Розвинути в ряд Тейлора}
\[
\sin\frac{\pi x}{4}, \quad x_0=2.
\]

Запишемо формулу Тейлора:
\[
f(x)=f(x_0)+f'(x_0)(x-x_0)+\frac{f''(x_0)}{2!}(x-x_0)^2+\dots
\]

Маємо:
\[
f(x)=\sin\frac{\pi x}{4},
\quad
f'(x)=\frac{\pi}{4}\cos\frac{\pi x}{4},
\quad
f''(x)=-\frac{\pi^2}{16}\sin\frac{\pi x}{4}.
\]

Обчислюємо значення в точці \(x_0=2\):
\[
f(2)=\sin\frac{\pi}{2}=1,
\]
\[
f'(2)=\frac{\pi}{4}\cos\frac{\pi}{2}=0,
\]
\[
f''(2)=-\frac{\pi^2}{16}\sin\frac{\pi}{2}
=-\frac{\pi^2}{16}.
\]

Підставляємо в ряд Тейлора:
\[
\sin\frac{\pi x}{4}
=
1
-\frac{\pi^2}{32}(x-2)^2
+\dots
\]

\textbf{Задача №9. Відновити аналітичну функцію \(f(z)\), якщо}
\[
\Re f(z)=u(x,y)=\frac{x}{x^2+y^2}, \qquad f(1)=1.
\]

Нехай
\[
f(z)=u(x,y)+iv(x,y),
\quad z=x+iy.
\]

\textbf{1. Обчислимо частинні похідні функції \(u(x,y)\).}

\[
u_x=\frac{\partial}{\partial x}\!\left(\frac{x}{x^2+y^2}\right)
=\frac{(x^2+y^2)-2x^2}{(x^2+y^2)^2}
=\frac{y^2-x^2}{(x^2+y^2)^2},
\]

\[
u_y=\frac{\partial}{\partial y}\!\left(\frac{x}{x^2+y^2}\right)
=-\frac{2xy}{(x^2+y^2)^2}.
\]

\textbf{2. Запишемо рівняння Коші–Рімана.}

\[
\begin{cases}
v_y=u_x,\\
v_x=-u_y.
\end{cases}
\]

Отже,
\[
v_x=\frac{2xy}{(x^2+y^2)^2}.
\]

\textbf{3. Знайдемо функцію \(v(x,y)\) інтегруванням за \(x\).}

\[
v(x,y)=\int \frac{2xy}{(x^2+y^2)^2}\,dx.
\]

Покладемо \(t=x^2+y^2\), тоді \(dt=2x\,dx\):
\[
v(x,y)=\int \frac{y}{t^2}\,dt
=-\frac{y}{x^2+y^2}+\varphi(y),
\]
де \(\varphi(y)\) — функція лише від \(y\).

\textbf{4. Знайдемо \(\varphi(y)\) з другого рівняння Коші–Рімана.}

\[
v_y=\frac{\partial}{\partial y}
\left(-\frac{y}{x^2+y^2}+\varphi(y)\right)
=\frac{y^2-x^2}{(x^2+y^2)^2}+\varphi'(y).
\]

За умовою \(v_y=u_x\), отже:
\[
\varphi'(y)=0 \quad \Rightarrow \quad \varphi(y)=C.
\]

\textbf{5. Запишемо аналітичну функцію.}

\[
f(z)=u+iv
=\frac{x}{x^2+y^2}-i\frac{y}{x^2+y^2}+iC
=\frac{x-iy}{x^2+y^2}+iC.
\]

\textbf{6. Використаємо умову \(f(1)=1\).}

Для \(z=1\): \(x=1,\ y=0\):
\[
f(1)=1+iC=1 \quad \Rightarrow \quad C=0.
\]

\textbf{Остаточний результат:}
\[
\boxed{f(z)=\frac{1}{z}}.
\]

\textbf{Задача 10.1. Обчислити інтеграл}
\[
\int_L z\overline{z}\,dz,
\]
де контур \(L\) задано рівнянням
\[
y=1-2x,\quad x:\ i\to \frac12.
\]

\textbf{Параметризація контуру.}

Нехай \(x=t\), тоді:
\[
y=1-2t,\quad t\in[0,\tfrac12].
\]

Комплексна змінна:
\[
z(t)=t+i(1-2t),
\quad
\overline{z}(t)=t-i(1-2t).
\]

Диференціал:
\[
dz=(1-2i)\,dt.
\]

\textbf{Підінтегральний вираз.}

\[
z\overline{z}
=
t^2+(1-2t)^2
=
5t^2-4t+1.
\]

\textbf{Інтегрування.}

\[
\int_L z\overline{z}\,dz
=
\int_0^{1/2}(5t^2-4t+1)(1-2i)\,dt.
\]

\[
=(1-2i)\int_0^{1/2}(5t^2-4t+1)\,dt.
\]

\[
\int_0^{1/2}(5t^2-4t+1)\,dt
=
\left[\frac{5}{3}t^3-2t^2+t\right]_0^{1/2}
=
\frac{5}{24}.
\]

\[
\boxed{\int_L z\overline{z}\,dz=\frac{5}{24}(1-2i)}.
\]

\textbf{Задача 11.1. Знайти всі лоранові розвинення функції}
\[
f(z)=\frac{3z-36}{z^4+3z^3-18z^2},
\quad z_0=0.
\]

Розкладемо знаменник:
\[
z^4+3z^3-18z^2=z^2(z-3)(z+6).
\]

Подамо функцію у вигляді простих дробів:
\[
\frac{3z-36}{z^2(z-3)(z+6)}
=
\frac{A}{z}+\frac{B}{z^2}+\frac{C}{z-3}+\frac{D}{z+6}.
\]

Звідси:
\[
A=-\frac12,\quad B=2,\quad C=\frac13,\quad D=\frac16.
\]

Отже:
\[
f(z)=\frac{2}{z^2}-\frac{1}{2z}
+\frac{1/3}{z-3}+\frac{1/6}{z+6}.
\]

\textbf{Розвинення при \(|z|<3\):}

\[
\frac{1}{z-3}
=-\frac13\sum_{n=0}^\infty\left(\frac{z}{3}\right)^n,
\quad
\frac{1}{z+6}
=\frac16\sum_{n=0}^\infty\left(-\frac{z}{6}\right)^n.
\]

Отримано лорановий ряд у кільці \(0<|z|<3\).

\textbf{Задача 12.2. Визначити тип особливих точок}
\[
f(z)=\frac{z^2+1}{(z-i)^2(z^2+4)}.
\]

Особливі точки:
\[
z=i,\quad z=\pm2i.
\]

\textbf{Точка \(z=i\).}

Чисельник:
\[
z^2+1\big|_{z=i}=0,
\]
нуль першого порядку.

Знаменник має множник \((z-i)^2\),
тобто нуль другого порядку.

Після скорочення:
\[
f(z)\sim\frac{1}{z-i}.
\]

\[
\boxed{z=i\text{ — полюс першого порядку}}.
\]

\textbf{Точки \(z=\pm2i\).}

Чисельник \(\neq0\), знаменник має нуль першого порядку.

\[
\boxed{z=\pm2i\text{ — прості полюси}}.
\]

\textbf{Задача 13.2. Обчислити інтеграл}
\[
\oint_{|z|=2}\frac{1-\cos z^2}{z^2}\,dz.
\]

Розкладемо чисельник у ряд Тейлора:
\[
\cos z^2
=
1-\frac{z^4}{2!}+\frac{z^8}{4!}-\dots
\]

\[
1-\cos z^2
=
\frac{z^4}{2!}-\frac{z^8}{4!}+\dots
\]

Тоді:
\[
\frac{1-\cos z^2}{z^2}
=
\frac{z^2}{2}-\frac{z^6}{24}+\dots
\]

У лорановому розкладі відсутній член \(\frac{1}{z}\).

За теоремою Коші:
\[
\oint_{|z|=2}\frac{1-\cos z^2}{z^2}\,dz=0.
\]

\boxed{0}
\]
