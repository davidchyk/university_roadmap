# ЛР №1 — Командний рядок Linux (універсальні команди для всіх варіантів)

**Плейсхолдери (заміни на свої з таблиці варіантів):**\
`[A]` — команда
`[B]` — шлях каталогу
`[C]` — змінна середовища
`[D]` — ім’я змінної
`[E]` — значення змінної

```bash
# 0) Запуск термінала: Ctrl+Alt+T

# 1) Версія ядра, архітектура, дистрибутив
uname -r
uname -m
cat /etc/os-release

# 2.1) Довідка по команді [A]
man [A]

# 2.2) Перехід до адміністратора
sudo -i
whoami # Хто я?

# 2.3) Перейти в каталог [B]
cd [B]
pwd

# 2.4) Перейти в домашній каталог
cd ~

# 2.5) Вивести всі змінні середовища
printenv
# або:
env

# 2.6) Вивести змінну [C]
echo "$[C]"

# 2.7) Встановити змінну [D] зі значенням [E]
export [D]="[E]" #або export D=E

# 2.8) Вивести змінну [D]
echo "$[D]"

# 2.9) Вивести всі змінні середовища (ще раз)
printenv

# 2.10) Видалити змінну [D]
unset [D]

# 2.11) Показати історію команд
history

# 2.12) Повторно встановити змінну [D] через history
!export
# або:
!<номер>

# 2.13) Вивести змінну [D]
echo "$[D]"

# 2.14) Вийти з адміністратора
exit

# 2.15) Вивести змінну [D] (у звичайному сеансі)
echo "$[D]"
```
