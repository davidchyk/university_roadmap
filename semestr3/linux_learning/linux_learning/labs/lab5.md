# ЛР №5 — Управління процесами (усі варіанти)
# Підготовка (виконати один раз перед варіантами)
- запустити довгий процес і запам’ятати PID:
```bash
sleep 1000 &           # у фоні
echo $!                # PID
```
#   - перевести у фон (якщо запускали у форграунді): Ctrl+Z → bg
#   - дізнатися PPID:
```bash
ps -o ppid= -p <PID>
```
#   - переглянути дерево процесів:
```bash
pstree -ap | grep <PID>
```
#   - знайти процес у top:
```bash
top  # (натисни L, введи PID для пошуку; q — вийти)
```
#   - завершити процес:
```bash
kill <PID>             # SIGTERM
kill -9 <PID>          # SIGKILL (якщо не завершився коректно)
```

# ========================= ВАРІАНТ 5 =========================
```bash
# Імена користувачів, під якими є процеси:
ps -eo user= | sort -u

# Завершити всі процеси певного ЗВИЧАЙНОГО користувача (приклад: user1)
U=user1
pgrep -u "$U" -l            # перевірити, що буде завершено
pkill -u "$U"               # м’яко (SIGTERM)
sleep 2
pgrep -u "$U" >/dev/null && pkill -KILL -u "$U"  # примусово, якщо лишились
```

# ========================= ВАРІАНТ 10 =========================
``` bash

# TOP — налаштувати колонки у порядку:
#   USER | PID | %CPU | %MEM | COMMAND
# Інтерактивно в top:
#   1) Запустити: top
#   2) 'f' (Fields Management) → пробілом лишити USER, PID, %CPU, %MEM, COMMAND
#   3) '>' / '<' — змінити порядок колонок (USER перша, далі PID, %CPU, %MEM, COMMAND)
#   4) 'W' — зберегти конфіг (у ~/.config/procps/toprc), 'q' — вийти
#
# Перевірочний аналог через ps:
ps -eo user,pid,pcpu,pmem,cmd --sort=-pcpu
```

#   1) top → 'f' → лишити PID, PPID, PR, VIRT, COMMAND
#   2) '>' / '<' розставити у вказаному порядку → 'W' зберегти
#
# Перевірочний аналог через ps:

# ========================= ВАРІАНТ 13 =========================
```bash

# НЕ root:
sleep 1000 &            # запустити щось у фоні
pid=$!
renice +5  -p "$pid"    # зменшити пріоритет (підвищити nice)
renice -5  -p "$pid"    # СКОРІШ за все ВІДМОВА (підвищення пріоритету без sudo)
kill "$pid"             # завершити

# ПІД root (sudo -i або перед командами додати sudo):
sleep 1000 & pid=$!
renice +5  -p "$pid"
renice -5  -p "$pid"    # тепер дозволено знижувати nice (підвищувати пріоритет)
kill "$pid"
```
